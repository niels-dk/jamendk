# merelyadream.com
#
# This file is a request, not a wall. Well-behaved crawlers honour it; the
# scanners hammering /login and /register do not, and nothing here is expected
# to slow them down. The real protection for private URLs is the
# "X-Robots-Tag: noindex" header index.php sends on every route that is not a
# public page — robots.txt cannot stop a URL from being indexed if someone
# links to it from elsewhere.
#
# What this DOES do is keep Google's crawl budget on the pages meant to be
# found, instead of spending it on app routes that only answer to a login.

User-agent: *

# Share links. The token in the URL IS the credential, so these must never be
# crawled or indexed. Listed here as well as noindexed, belt and braces.
Disallow: /t/
Disallow: /trips/
Disallow: /documents/

# One-time token URLs from email. Nothing to index, and fetching one would
# spend a link that belongs to a person.
Disallow: /verify/
Disallow: /verify-resend
Disallow: /reset/
Disallow: /forgot

# The application itself. Everything below answers only to a signed-in user,
# so a crawler gets a redirect and learns nothing.
Disallow: /dashboard
Disallow: /account
Disallow: /admin
Disallow: /teams
Disallow: /capture
Disallow: /dreams
Disallow: /visions
Disallow: /moods
Disallow: /api/

# /login and /register are deliberately NOT listed, and must not be re-added.
#
# Disallow and noindex work against each other. A crawler has to FETCH a page
# to read its X-Robots-Tag, so forbidding the fetch means the noindex is never
# seen — and a blocked URL that something links to can sit in the index as a
# bare URL forever. Google even has a name for it: "Indexed, though blocked by
# robots.txt".
#
# That is what happened here. Six /login?next=/dreams/... and
# /register?next=/dreams/... URLs were filed as "duplicate without canonical",
# and blocking them left Google unable to re-crawl and clear them. Letting it
# fetch means it finds both signals we already send: a canonical pointing at
# the bare path, and noindex.
#
# The rules above are the opposite case on purpose. For /t/ and /documents/ the
# fetch IS the harm — the token in the URL is the credential — so there we
# forbid crawling and accept that our noindex may never be read.
